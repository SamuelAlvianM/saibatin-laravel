# =============================================================================
#  .htaccess untuk public_html — SAIBATIN Laravel di cPanel
# =============================================================================
#
#  Salin isi berkas ini menjadi  /home/<akun>/public_html/.htaccess
#
#  Gabungan tiga hal yang selama ini terpisah:
#    1. aturan rewrite Laravel  (dari public/.htaccess bawaan project)
#    2. redirect HTTPS          (dari .htaccess lama document root)
#    3. blok php_value + handler cPanel (dari .htaccess lama document root)
#
#  🔴 Blok nomor 3 DIHASILKAN cPanel. Kalau kelak Anda mengubah setelan lewat
#     MultiPHP INI Editor atau MultiPHP Manager, cPanel akan MENULIS ULANG blok
#     itu di berkas ini dan menghapus perubahan manual di dalamnya. Karena itu
#     blok Laravel diletakkan DI ATAS — supaya yang ditimpa cPanel hanya
#     bagiannya sendiri di bawah.
#
#  ⚠️ Dua nilai sengaja dinaikkan dari .htaccess lama, dengan alasannya:
#       memory_limit  128M → 512M
#       max_execution_time 90 → 180
#     Keduanya untuk ekspor Excel statistik (± 170 MB, 13 sheet). Kalau hosting
#     menolak (LVE atau php_admin_value), yang gagal HANYA ekspor Excel penuh
#     dan lampiran gambar di PDF — portalnya sendiri tetap jalan.
#
# =============================================================================

<IfModule mod_rewrite.c>
    <IfModule mod_negotiation.c>
        Options -MultiViews -Indexes
    </IfModule>

    RewriteEngine On

    # ── HTTPS ────────────────────────────────────────────────────────────────
    # Dipertahankan dari .htaccess lama. Wajib: cookie sesi portal ini dikirim
    # dengan Secure, jadi di http:// login akan gagal tanpa pesan.
    RewriteCond %{HTTPS} !=on
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

    # ── Header Authorization ─────────────────────────────────────────────────
    RewriteCond %{HTTP:Authorization} .
    RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

    # ── Header X-XSRF-Token ──────────────────────────────────────────────────
    # 🔴 ADA di public/.htaccess Laravel tapi TIDAK ada di .htaccess lama Anda.
    # Tanpa baris ini seluruh POST dari dashboard & formulir dijawab 419:
    # portal ini mengirim token lewat header X-XSRF-TOKEN (HANDOFF §5 no. 9),
    # dan sebagian konfigurasi Apache membuang header non-standar sebelum PHP
    # sempat melihatnya.
    RewriteCond %{HTTP:x-xsrf-token} .
    RewriteRule .* - [E=HTTP_X_XSRF_TOKEN:%{HTTP:X-XSRF-Token}]

    # ── Slash di akhir ───────────────────────────────────────────────────────
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteCond %{REQUEST_URI} (.+)/$
    RewriteRule ^ %1 [L,R=301]

    # ── Teruskan ke front controller ─────────────────────────────────────────
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteRule ^ index.php [L]
</IfModule>

# ── Jaring pengaman: jangan sajikan berkas rahasia ───────────────────────────
# Aplikasi memang ditaruh DI LUAR public_html, jadi berkas-berkas ini
# seharusnya tidak ada di sini. Blok ini untuk kalau suatu saat ada yang
# menyalin seluruh folder project ke public_html karena terburu-buru.
<FilesMatch "^(\.env.*|\.git.*|composer\.(json|lock)|package(-lock)?\.json|artisan|phpunit\.xml|vite\.config\.js)$">
    Require all denied
</FilesMatch>

# Mesin OCR & aset build boleh di-cache lama — namanya sudah ber-hash.
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType application/javascript "access plus 1 year"
    ExpiresByType text/css "access plus 1 year"
    ExpiresByType application/wasm "access plus 1 year"
    ExpiresByType image/png "access plus 1 month"
    ExpiresByType image/jpeg "access plus 1 month"
    ExpiresByType image/webp "access plus 1 month"
</IfModule>

# pertama dibuat — termasuk `AddHandler application/x-httpd-ea-php82 .php`.
# Blok itu MILIK SERVER, bukan milik aplikasi: ia menentukan PHP versi berapa
# yang menjalankan berkas .php di host tersebut.
#
# Terukur 13 Sep 2026 di server SIMPONI Halmahera Selatan. `.htaccess` paket
# ditambahkan di bawah blok milik server, sehingga AddHandler ea-php82 datang
# BELAKANGAN dan menang atas ea-php83 milik server. Akibatnya web dijalankan
# PHP 8.2, `vendor/composer/platform_check.php` menuntut >= 8.3, dan SETIAP
# halaman menjawab 500 — sementara `php artisan` tetap mulus karena CLI
# memakai PHP 8.3. Dua jam tersita untuk menemukannya.
#
# Blok milik server sudah ada di `public_html/.htaccess` sejak akun dibuat.
# Yang benar adalah MENAMBAHKAN aturan portal ini di bawahnya, bukan membawa
# serta blok versi PHP dari mesin lain.
#
# Batas PHP (memory_limit, max_execution_time) diatur lewat cPanel → MultiPHP
# INI Editor, yang menuliskannya ke `.user.ini` — dan pada host ber-PHP-FPM
# atau lsapi, `.user.ini` itulah yang dipatuhi, bukan php_value di sini.
